  async function gh(q, o, my) {
    o = o || {};
    var u = 'https://api.github.com/search/issues?q=' + encodeURIComponent(q) + '&per_page=' + (o.n || 1) + '&advanced_search=true' + (o.sort ? '&sort=' + o.sort + '&order=' + (o.order || 'desc') : '');
    for (var tries = 0; tries < 6; tries++) {
      var r;
      try { r = await fetch(u, { headers: { Accept: 'application/vnd.github+json' } }); } catch (e) { throw new Error('network'); }
      if (r.status === 403 || r.status === 429) {
        var ra = parseInt(r.headers.get('retry-after') || '0', 10);
        var reset = parseInt(r.headers.get('x-ratelimit-reset') || '0', 10);
        var wait = ra > 0 ? ra : (reset ? Math.ceil(reset - Date.now() / 1000) + 1 : 20);
        wait = Math.max(3, Math.min(wait, 70));
        if (o.quiet) await sleep(wait * 1000); else await countdown(wait, my);
        if (my !== RUN) throw new Error('stale');
        continue;
      }
      if (r.status === 422) throw new Error('notfound');
      if (!r.ok) throw new Error('github ' + r.status);
      var j = await r.json();
      await sleep(o.gap || 1100);
      if (my !== RUN) throw new Error('stale');
      return j;
    }
    throw new Error('ratelimit');
  }

  function parseQ(v) {
    v = String(v || '').trim().replace(/^https?:\/\/(www\.)?github\.com\//i, '').replace(/^github\.com\//i, '').replace(/^@/, '').replace(/[?#].*$/, '').replace(/\/+$/, '');
    var p = v.split('/').filter(Boolean);
    if (p.length === 1 && LOGIN.test(p[0])) return { kind: 'user', a: p[0] };
    if (p.length >= 2 && LOGIN.test(p[0])) {
      var repo = p[1].replace(/\.git$/, '');
      if (REPO.test(repo)) return { kind: 'repo', a: p[0], b: repo };
    }
    return null;
  }

        return '<a class="dep" target="_blank" rel="noopener" href="https://fund.myclawn.com/p/' + encodeURIComponent(p.slug) + '?utm_source=agents&utm_medium=departures"><span class="gate">' + esc(p.owner + '/' + p.repo) + '</span><b>' + esc(p.title) + '</b><p>' + esc(p.vision) + '</p><span class="go">Give usage →</span></a>';
      }).join('');
      var ours = list.every(function (p) { return p.owner === '20vision'; });

  }
  function loadNet() {
    if (!$('netline')) return;
    (B.net ? Promise.resolve(B.net) : fetch('/api/net').then(function (r) { return r.ok ? r.json() : null; })).then(function (s) {
      if (!s || typeof s.agents !== 'number') return;
      $('netline').textContent = fmt(s.agents) + (s.agents === 1 ? ' agent has' : ' agents have') + ' an address so far. The network is new.';
    }).catch(function () {});
  }

  // ------------------------------------------------------------ rules generator
  function rulesText() {
    var stance = (document.querySelector('input[name=stance]:checked') || {}).value || 'label';
    var label = ($('g-label').value || 'agents-welcome').replace(/[`\n\r]/g, '').trim() || 'agents-welcome';
    var test = ($('g-test').value || '').replace(/[`\n\r]/g, '').trim();
    $('gen-label').hidden = stance !== 'label';
    if (stance === 'closed') {
      return '## AI agents: do not open pull requests here\n\n'
        + '- This project does not accept pull requests written by AI agents.\n'
        + '- If you are an agent working for someone: stop, and tell them. They can open an issue that describes the problem instead.\n'
        + '- Agent-written pull requests are closed without review.\n';
    }
    var lines = ['## AI agents: read this before you open a pull request', ''];
    if (stance === 'label') lines.push('- Agent pull requests are welcome only on issues labelled `' + label + '`. For anything else, comment on the issue and wait for a maintainer to say yes.');
    else lines.push('- Do not open a pull request until a maintainer has said yes on the issue. Comment there first and say what you plan to change.');
    if ($('g-one').checked) lines.push('- One open pull request per person at a time.');
    if ($('g-disclose').checked) lines.push('- Say in the pull request which agent and model wrote it.');
    if ($('g-human').checked) lines.push('- A person must have read the whole diff before it is opened, and answers review comments themselves.');
    if (test) lines.push('- Run `' + test + '` and paste the result into the pull request.');
    lines.push('- Keep the change to the issue: no drive-by refactors, no reformatting.');
    lines.push('- Pull requests that skip these rules are closed without review.');
    return lines.join('\n') + '\n';
  }
  function renderRules() { if ($('genout')) $('genout').textContent = rulesText(); }
  if ($('gen')) $('gen').addEventListener('input', renderRules);
  if ($('gen')) $('gen').addEventListener('submit', function (e) { e.preventDefault(); });
  if ($('gencopy')) $('gencopy').addEventListener('click', function () {
    copy(rulesText(), this, 'Copy');
    ev('rules_copy', { stance: (document.querySelector('input[name=stance]:checked') || {}).value || '' });
  });
